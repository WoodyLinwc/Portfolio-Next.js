Array Operations

Array Creation and Basic Operations

// Creating arrays
const arr1 = [1, 2, 3];
const arr2 = new Array(3);              // [empty × 3]
const arr3 = Array.of(1, 2, 3);         // [1, 2, 3]
const arr4 = Array.from({length: 3}, (_, i) => i); // [0, 1, 2]

// Adding/removing elements
arr.push(4);        // add to end, returns new length
arr.pop();          // remove from end, returns removed element
arr.unshift(0);     // add to beginning, returns new length
arr.shift();        // remove from beginning, returns removed element

Array Transformation Methods (Detailed)

.map(callback) - Creates a new array by transforming each element 
// Callback signature: (value, index, array)
const numbers = [1, 2, 3, 4];

// Basic usage
const doubled = numbers.map(x => x * 2); // [2, 4, 6, 8]

// Using all parameters
const withIndex = numbers.map((value, index, array) => ({
  value,
  index,
  isLast: index === array.length - 1
}));
// [{value: 1, index: 0, isLast: false}, ...]

// Chaining methods
const result = numbers
  .map(x => x * 2)
  .filter(x => x > 4)
  .map(x => `Number: ${x}`); // ["Number: 6", "Number: 8"]
.filter(callback) - Creates a new array with elements that pass the test 
const numbers = [1, 2, 3, 4, 5, 6];

// Basic filtering
const evens = numbers.filter(x => x % 2 === 0); // [2, 4, 6]

// Using index parameter
const evenPositions = numbers.filter((value, index) => index % 2 === 0);
// [1, 3, 5] (elements at even indices)

// Complex filtering with chaining
const filtered = numbers
  .filter(x => x > 2)           // [3, 4, 5, 6]
  .map(x => x * x)              // [9, 16, 25, 36]
  .filter(x => x < 30);         // [9, 16, 25]
Key Differences:
• .map() always returns an array of the same length
• .filter() returns an array with potentially fewer elements
• .forEach() returns undefined, used for side effects only
• All create new arrays (immutable operations)

Other Array Methods

const arr = [1, 2, 3, 4, 5];

// .reduce() - Reduces array to single value
const sum = arr.reduce((acc, val, index, array) => acc + val, 0); // 15
const max = arr.reduce((max, val) => val > max ? val : max); // 5

// .find() and .findIndex()
const found = arr.find(x => x > 3);      // 4 (first match)
const index = arr.findIndex(x => x > 3); // 3 (index of first match)

// .some() and .every()
arr.some(x => x > 4);    // true (at least one > 4)
arr.every(x => x > 0);   // true (all > 0)

// .forEach() - Just loops, returns undefined
arr.forEach((value, index, array) => {
  console.log(`${index}: ${value}`);
  // Cannot break or return early
});

Object Operations

Object Creation and Property Access

const obj = {a: 1, b: 2, c: 3};

// Property access methods
obj.a;          // 1 (dot notation)
obj['a'];       // 1 (bracket notation)
obj?.d?.e;      // undefined (optional chaining)

// Dynamic property names
const key = 'dynamicKey';
const obj2 = {
  [key]: 'value',           // computed property
  [`${key}2`]: 'value2'     // template literal key
};

Object Iteration and Transformation

const obj = {name: 'John', age: 30, city: 'NYC'};

// Object.entries() with destructuring
Object.entries(obj).forEach(([key, value]) => {
  console.log(`${key}: ${value}`);
});

// Transform object to array
const keyValuePairs = Object.entries(obj)
  .map(([key, value]) => `${key}=${value}`)
  .join('&'); // "name=John&age=30&city=NYC"

// Filter object properties
const filtered = Object.fromEntries(
  Object.entries(obj).filter(([key, value]) => typeof value === 'string')
); // {name: 'John', city: 'NYC'}

String Operations

Advanced String Methods

const str = "  Hello World  ";

// Template literals with expressions
const name = "John";
const age = 30;
const greeting = `Hello ${name}, you are ${age} years old.`;

// String methods chaining
const processed = str
  .trim()                    // "Hello World"
  .toLowerCase()             // "hello world"
  .split(' ')                // ["hello", "world"]
  .map(word => word.charAt(0).toUpperCase() + word.slice(1))
  .join(' ');                // "Hello World"

// Modern string methods
str.startsWith('  Hello');   // true
str.endsWith('World  ');     // true
str.includes('World');       // true
'abc'.repeat(3);             // "abcabcabc"
'5'.padStart(3, '0');        // "005"

Number Operations

Number Validation and Conversion

// Safe number operations
Number.isInteger(42);        // true
Number.isFinite(42);         // true
Number.isNaN(NaN);           // true
Number.isSafeInteger(42);    // true

// Parse vs Convert
parseInt('42px');            // 42
parseFloat('3.14abc');       // 3.14
Number('42px');              // NaN
+'42';                       // 42 (unary plus)

// Formatting numbers
const num = 1234.5678;
num.toFixed(2);              // "1234.57"
num.toPrecision(4);          // "1235"
num.toLocaleString();        // "1,234.568" (locale-dependent)

Date Operations

Date Creation and Manipulation

// Modern date handling
const now = new Date();
const specific = new Date('2023-12-25T10:30:00');
const fromParts = new Date(2023, 11, 25, 10, 30); // month is 0-indexed

// Date calculations
const tomorrow = new Date(now);
tomorrow.setDate(now.getDate() + 1);

const weekAgo = new Date(now.getTime() - 7 * 24 * 60 * 60 * 1000);

// Formatting dates
now.toISOString();           // "2023-12-25T15:30:00.000Z"
now.toLocaleDateString();    // "12/25/2023"
now.toLocaleTimeString();    // "3:30:00 PM"
now.toLocaleString('en-GB'); // "25/12/2023, 15:30:00"

Set & Map Operations

Set Operations (Advanced)

const set1 = new Set([1, 2, 3, 4]);
const set2 = new Set([3, 4, 5, 6]);

// Set operations
const union = new Set([...set1, ...set2]);           // {1,2,3,4,5,6}
const intersection = new Set([...set1].filter(x => set2.has(x))); // {3,4}
const difference = new Set([...set1].filter(x => !set2.has(x)));  // {1,2}

// Practical use case: Remove duplicates from array
const arr = [1, 2, 2, 3, 3, 4];
const unique = [...new Set(arr)]; // [1, 2, 3, 4]

Map Operations (Advanced)

const map = new Map();

// Maps can use any type as key
map.set('string', 'value1');
map.set(42, 'value2');
map.set(true, 'value3');
map.set({}, 'value4');

// Iterate over Map
map.forEach((value, key) => console.log(key, value));

// Convert Map to Object (string keys only)
const obj = Object.fromEntries(map);

// WeakMap for memory-efficient object associations
const weakMap = new WeakMap();
const element = document.getElementById('myElement');
weakMap.set(element, {data: 'associated data'});

RegExp Operations

Regular Expression Patterns

// Common regex patterns
const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
const phoneRegex = /^\(\d{3}\)\s\d{3}-\d{4}$/; // (555) 123-4567
const urlRegex = /https?:\/\/[^\s]+/g;

// Using regex with string methods
const text = "Contact us at john@example.com or call (555) 123-4567";
const emails = text.match(emailRegex);
const cleaned = text.replace(phoneRegex, '[PHONE]');
const parts = text.split(/\s+/); // Split by whitespace

// Named capture groups (modern)
const dateRegex = /(?\d{4})-(?\d{2})-(?\d{2})/;
const match = '2023-12-25'.match(dateRegex);
console.log(match.groups.year); // "2023"

Async Operations

Promises and Async/Await

// Promise creation and chaining
const fetchUserData = (id) => {
  return fetch(`/api/users/${id}`)
    .then(response => response.json())
    .then(data => ({
      ...data,
      fullName: `${data.firstName} ${data.lastName}`
    }))
    .catch(error => {
      console.error('Failed to fetch user:', error);
      return null;
    });
};

// Async/await equivalent
const fetchUserDataAsync = async (id) => {
  try {
    const response = await fetch(`/api/users/${id}`);
    const data = await response.json();
    return {
      ...data,
      fullName: `${data.firstName} ${data.lastName}`
    };
  } catch (error) {
    console.error('Failed to fetch user:', error);
    return null;
  }
};

// Promise utilities
Promise.all([promise1, promise2, promise3])      // Wait for all
  .then(results => console.log(results));

Promise.allSettled([promise1, promise2])         // Don't fail fast
  .then(results => {
    results.forEach(result => {
      if (result.status === 'fulfilled') {
        console.log('Success:', result.value);
      } else {
        console.log('Error:', result.reason);
      }
    });
  });

Web APIs Examples

localStorage.setItem('user', JSON.stringify({name: 'John', age: 30}));

// Fetch API
const fetchData = async (url) => {

    method: 'POST',
    headers: {

    },
    body: JSON.stringify({key: 'value'})

const observer = new IntersectionObserver((entries) => {
  entries.forEach(entry => {
    if (entry.isIntersecting) {
      entry.target.src = entry.target.dataset.src;
      observer.unobserve(entry.target);
    }
  });
});

document.querySelectorAll('img[data-src]').forEach(img => {
  observer.observe(img);

Utility Functions

Advanced Utility Patterns

// Debounce - Delay execution until after calls have stopped
const debounce = (func, delay) => {
  let timeoutId;
  return (...args) => {
    clearTimeout(timeoutId);
    timeoutId = setTimeout(() => func.apply(this, args), delay);
  };
};

// Usage: Debounce search input
const searchInput = document.getElementById('search');
const debouncedSearch = debounce((query) => {
  console.log('Searching for:', query);
}, 300);

searchInput.addEventListener('input', (e) => {
  debouncedSearch(e.target.value);
});

// Throttle - Limit execution frequency
const throttle = (func, limit) => {
  let inThrottle;
  return (...args) => {
    if (!inThrottle) {
      func.apply(this, args);
      inThrottle = true;
      setTimeout(() => inThrottle = false, limit);
    }
  };
};

// Deep clone (handles nested objects/arrays)
const deepClone = (obj) => {
  if (obj === null || typeof obj !== 'object') return obj;
  if (obj instanceof Date) return new Date(obj.getTime());
  if (obj instanceof Array) return obj.map(deepClone);
  if (typeof obj === 'object') {
    const clonedObj = {};
    for (const key in obj) {
      if (obj.hasOwnProperty(key)) {
        clonedObj[key] = deepClone(obj[key]);
      }
    }
    return clonedObj;
  }
};

// Curry function
const curry = (fn) => {
  return function curried(...args) {
    if (args.length >= fn.length) {
      return fn.apply(this, args);
    } else {
      return (...moreArgs) => curried(...args, ...moreArgs);
    }
  };
};

// Usage
const multiply = curry((a, b, c) => a * b * c);
const multiplyByTwo = multiply(2);
const multiplyByTwoAndThree = multiplyByTwo(3);
console.log(multiplyByTwoAndThree(4)); // 24

Underscore (_) Usage in JavaScript

The underscore character (_) has several conventional uses in JavaScript:

1. Unused Function Parameters

// When you need to access the third parameter but not the first two
array.map((_, index) => index);  // Get array indices
array.forEach((_, i, arr) => console.log(arr.length));

// In destructuring when you don't need certain values
const [first, _, third] = [1, 2, 3];  // Skip second element
const {name, age: _, ...rest} = person;  // Skip age property

2. Private/Internal Variables Convention

class MyClass {
  constructor() {
    this._privateVar = 'internal use';  // Convention: private
    this.__reallyPrivate = 'very internal';  // Convention: very private
  }
  
  _internalMethod() {  // Convention: internal method
    return this._privateVar;
  }
}

// Module pattern
const myModule = (() => {
  let _privateCounter = 0;  // Private variable
  
  return {
    increment: () => ++_privateCounter,
    getCount: () => _privateCounter
  };
})();

3. Lodash/Underscore.js Library

// If using Lodash library (commonly aliased as _)
const _ = require('lodash');

const users = [
  {name: 'John', age: 30},
  {name: 'Jane', age: 25},
  {name: 'Bob', age: 35}
];

// Lodash utility functions
const sortedNames = _.sortBy(users, 'age').map(u => u.name);
const grouped = _.groupBy(users, u => u.age > 30 ? 'senior' : 'junior');
const debounced = _.debounce(myFunction, 300);

4. Placeholder in Template Literals

// When building dynamic strings
const buildUrl = (base, path, params = {}) => {
  const query = Object.entries(params)
    .map(([key, value]) => `${key}=${encodeURIComponent(value)}`)
    .join('&');
  
  return query ? `${base}${path}?${query}` : `${base}${path}`;
};

5. Previous Result Reference

// In REPL or console, _ often refers to the last result
// This is environment-specific (Node.js REPL, browser console)
console.log(42);  // 42
console.log(_ * 2);  // 84 (in some environments)
Best Practice: Use underscore prefixes for internal/private members, and single underscore for unused parameters to improve code readability.
Note: Underscore prefixes are conventions only - they don't provide actual privacy in JavaScript. Use private class fields (#property) for true privacy in modern JavaScript.
